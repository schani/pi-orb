import type { HistoryRecord } from "@pi-orb/protocol";
import { ConditionVariable, type SimulationTask } from "determined";
import { ok, okAsync, ResultAsync } from "neverthrow";
import { describe, expect, it } from "vitest";
import { makeHarness, makeOrbRow, makeProjectRow } from "../testkit/fixtures.ts";
import { LogCapture, runDst, waitUntil } from "../testkit/sim.ts";
import type { AgentPlane } from "./agent-ports.ts";
import { pollAllOnce } from "./loops.ts";

const phases = ["health", "claim", "delivery", "history", "commit", "naming", "catchup"] as const;

describe("real multi-orb polling (DST)", () => {
  for (const phase of phases) {
    it(`delivers B on later ticks while A holds ${phase}`, async () => {
      await runDst(
        { name: `poll-held-${phase}`, iterations: 20, lateTimerProbability: 0, logCapture: new LogCapture() },
        async (sim) => {
          const h = makeHarness({ constants: {
            reconcileTickMs: 20, historyPullIntervalMs: 20, runtimeRequestTimeoutMs: 100_000,
          } });
          h.store.seedProject(makeProjectRow("project"));
          for (const id of ["a", "b"])
            h.store.seedOrb(makeOrbRow(id, "project", "creating", { name: id === "a" && phase === "naming" ? null : id }));
          const gate = new ConditionVariable("held poll phase");
          let held = false;
          let released = false;
          let ready = false;
          const stop = new AbortController();
          const deliveries: string[] = [];
          const records = new Map<string, HistoryRecord[]>([["a", []], ["b", []]]);
          let batches = 0;
          async function hold(task: SimulationTask) {
            held = true;
            gate.notifyAll(task, "A held");
            while (!released) await gate.wait(task, "A release");
          }
          function record(task: SimulationTask, orbId: string, id: string, inbox = false): HistoryRecord {
            const list = records.get(orbId)!;
            return {
              id, parentId: list.at(-1)?.id ?? null, timestamp: new Date(task.wallNow()).toISOString(),
              overflow: {}, type: "message", role: "user", content: [{ type: "text", text: id }],
              ...(inbox ? { inboxMessageIds: [id] } : {}),
            };
          }
          const claim = h.store.claimNextOrbMessageBatch.bind(h.store);
          h.store.claimNextOrbMessageBatch = (task, request) => new ResultAsync((async () => {
            if (request.orbId === "a" && phase === "claim" && !released) await hold(task);
            return claim(task, request);
          })());
          const commit = h.store.commitPullBatch.bind(h.store);
          h.store.commitPullBatch = (task, request) => new ResultAsync((async () => {
            if (request.orbId === "a" && phase === "commit" && !released) await hold(task);
            return commit(task, request);
          })());
          const plane: AgentPlane = {
            placement: "central",
            health: (task, orb) => new ResultAsync((async () => {
              if (orb.id === "a" && phase === "health" && !released) await hold(task);
              return ok({ v: 1 as const, orbId: orb.id, runtimeInstanceId: orb.id,
                status: "ready" as const, checkoutCommit: "central", sessionId: orb.id, activity: "busy" as const });
            })()),
            deliverMessage: (task, orb, request) => new ResultAsync((async () => {
              if (orb.id === "a" && phase === "delivery" && !released) await hold(task);
              if (!deliveries.includes(request.messageId)) {
                records.get(orb.id)!.push(record(task, orb.id, request.messageId, true));
                deliveries.push(request.messageId);
              }
              gate.notifyAll(task, "delivered");
              return ok({ v: 1 as const, messageId: request.messageId, status: "persisted" as const,
                delivery: "steer" as const, operationId: "turn", duplicate: false });
            })()),
            pullHistory: (task, orb, request) => new ResultAsync((async () => {
              if (orb.id === "a" && phase === "history" && !released) await hold(task);
              const list = records.get(orb.id)!;
              if (orb.id === "a" && phase === "catchup" && !released) {
                list.push(record(task, orb.id, `catchup-${++batches}`));
                held = true;
                gate.notifyAll(task, "catchup continues");
                await task.sleep(5, "sustained new history");
              }
              const index = request.after === null ? -1 : list.findIndex((item) => item.id === request.after);
              const page = list.slice(index + 1);
              return ok({ v: 1 as const, orbId: orb.id, runtimeInstanceId: orb.id,
                activity: "busy" as const, session: { id: orb.id, overflow: {} }, records: page,
                cursor: page.at(-1)?.id ?? request.after, headId: list.at(-1)?.id ?? null });
            })()),
            prepareIdleStop: () => okAsync({ v: 1 as const, prepared: false }),
            suspend: () => okAsync(undefined), dispose: () => okAsync(undefined),
            close: () => okAsync(undefined), session: () => null,
          };
          const deps = { ...h.deps, agentPlane: plane,
            nameGenerator: { generate: (task: SimulationTask) => new ResultAsync((async () => {
              if (phase === "naming" && !released) await hold(task);
              return ok("A name");
            })()) },
          };
          const result = await sim.runTasks([
            { name: "poller", f: async (task) => {
              while (!ready) await gate.wait(task, "setup");
              while (!stop.signal.aborted) {
                await pollAllOnce(task, deps);
                await task.sleep(20, "next real poll sweep");
              }
            } },
            { name: "driver", f: async (task) => {
              records.get("a")!.push(record(task, "a", "seed"));
              expect((await h.store.enqueueOrbMessage(task, {
                orbId: "a", messageId: "a-input", content: [{ type: "text", text: "A input" }], now: task.wallNow(),
              })).isOk()).toBe(true);
              ready = true;
              gate.notifyAll(task, "setup ready");
              while (!held) await gate.wait(task, "A enters phase");
              for (const id of ["b-first", "b-later"]) {
                expect((await h.store.enqueueOrbMessage(task, {
                  orbId: "b", messageId: id, content: [{ type: "text", text: id }], now: task.wallNow(),
                })).isOk()).toBe(true);
                await waitUntil(task, `${id} receipt before A release`, () =>
                  h.store.messageSnapshots("b").some((row) => row.messageId === id && row.status === "delivered"),
                  { intervalMs: 20, timeoutMs: 1_000 });
                expect(released).toBe(false);
                await task.sleep(80, "future dispatcher ticks");
              }
              if (phase === "catchup") expect(batches).toBeGreaterThan(3);
              released = true;
              gate.notifyAll(task, "A released after B receipts");
              stop.abort();
            } },
          ]);
          expect(result.isOk(), result.isErr() ? result.error.message : "").toBe(true);
          expect(deliveries).toContain("b-later");
        },
      );
    });
  }
});
